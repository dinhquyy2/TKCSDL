\subsection{External Interfaces}
\begin{itemize}
    \item \textbf{Giao diện người dùng:} Admin Dashboard (desktop, tối ưu cho thao tác quản trị) và POS App dạng PWA (tối ưu cho tablet/thiết bị cầm tay, responsive).
    \item \textbf{Tích hợp bên ngoài:} Webhook Simulator mô phỏng payload từ Shopee/TikTok Shop/Lazada; kiến trúc cho phép thay thế bằng API thật của các sàn ở giai đoạn phát triển tiếp theo.
    \item \textbf{Thiết bị phần cứng:} Máy quét mã vạch (barcode scanner) qua giao diện bàn phím HID chuẩn tại POS; máy in hóa đơn qua Browser Print API.
\end{itemize}

\subsection{Quality Attributes}

\subsubsection{Performance}
\begin{itemize}
    \item API Response Time: tìm SKU/quét mã vạch tại POS p95 < 200ms; giao dịch tạo đơn/ghi < 500ms; báo cáo doanh thu tổng hợp < 2s với tập dữ liệu dưới 100.000 bản ghi.
    \item Concurrency \& Anti-Oversell: xử lý được kịch bản Flash Sale (50 đơn online đồng thời tranh 1 SKU còn 1 đơn vị tồn), bắt buộc dùng Pessimistic Locking hoặc Optimistic Locking để tuyệt đối không cho tồn kho khả dụng âm.
\end{itemize}

\subsubsection{Reliability \& Data Integrity}
\begin{itemize}
    \item Tuân thủ ACID: chuỗi Order Creation → Reserve → Deduct bắt buộc thực thi trong một giao dịch cơ sở dữ liệu nguyên tử, rollback toàn bộ nếu thất bại.
    \item Tính kiểm toán của Ledger: bảng InventoryLedger là append-only tuyệt đối, không cho phép UPDATE/DELETE; mọi điều chỉnh phải qua bút toán bù trừ.
    \item Mục tiêu uptime 99.5\% trong giờ hoạt động (7:00--22:00) cho hệ thống hoàn chỉnh.
\end{itemize}

\subsubsection{Security}
\begin{itemize}
    \item Mật khẩu người dùng băm bằng BCrypt/Argon2. Giao tiếp API qua HTTPS/TLS~1.3.
    \item JWT Access Token thời hạn ngắn (60 phút) kèm Refresh Token an toàn.
\end{itemize}

\subsubsection{Multi-tenancy \& Scalability}
\begin{itemize}
    \item Mô hình Shared Database, Separate Schema/TenantId. Mọi bảng nghiệp vụ đều có TenantId, bắt buộc dùng EF Core Global Query Filters để đảm bảo không rò rỉ dữ liệu giữa các tenant.
    \item Tối ưu index trên (TenantId, CreatedAt) và (TenantId, SKUId) cho bảng InventoryLedger và Orders để giữ hiệu năng truy vấn ở quy mô hàng triệu dòng ledger.
\end{itemize}

\subsubsection{Usability}
\begin{itemize}
    \item Giao diện POS tối giản, thanh toán dưới 3 lần click hoặc quét mã vạch + Enter.
    \item Responsive cho desktop, tablet, thiết bị POS cầm tay.
\end{itemize}

\subsubsection{Maintainability}
\begin{itemize}
    \item Tách biệt nghiêm ngặt theo Clean Architecture: Domain (COGS, Reserve), Application (use case), Infrastructure (EF Core, PostgreSQL, SignalR), Presentation (Web API).
    \item Các module lõi (Inventory Ledger, Anti-Oversell Reserve, COGS) phải đạt tối thiểu 80\% Unit \& Integration Test coverage khi được lập trình ở giai đoạn tiếp theo.
\end{itemize}
