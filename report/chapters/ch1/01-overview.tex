\section{Tổng quan dự án}
\subsection{Thông tin dự án}
\begin{itemize}
    \item \textbf{Tên dự án (Tiếng Anh):} Omnichannel Inventory and Sales Management System (OISM).
    \item \textbf{Tên dự án (Tiếng Việt):} Hệ thống quản lý bán hàng và tồn kho đa kênh.
    \item \textbf{Loại phần mềm:} Nền tảng SaaS đa khách thuê (multi-tenant), gồm Backend API, Admin Dashboard (Web) và ứng dụng POS dạng Progressive Web App (PWA).
    \item \textbf{Mục đích:} Xây dựng một hệ thống giúp doanh nghiệp bán lẻ quản lý tập trung tồn kho và đơn hàng đến từ nhiều kênh bán (cửa hàng vật lý, Shopee, TikTok Shop, Lazada), loại bỏ tình trạng bán vượt tồn kho (oversell), sai lệch giá vốn và mất khả năng truy vết dữ liệu.
    \item \textbf{Phạm vi bàn giao của nhóm:} Thiết kế hệ thống toàn diện (kiến trúc, đặc tả yêu cầu, mô hình dữ liệu, luồng nghiệp vụ, đặc tả API) và xây dựng cơ sở dữ liệu vật lý (SQL Server) kèm dữ liệu mẫu minh họa. Việc lập trình Backend API, Frontend, tích hợp AI, CI/CD và triển khai thực tế nằm ngoài phạm vi bàn giao của học kỳ này (xem chi tiết tại mục~\ref{sec:scope-limitations}).
\end{itemize}
